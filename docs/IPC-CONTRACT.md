# عقد الربط بين الواجهة و Rust (IPC Contract) — v1

> هذا العقد يتفق عليه طرفان يعملان بالتوازي: الواجهة (`apps/desktop/src`) والخلفية (`apps/desktop/src-tauri`).
> أي تغيير فيه يُسجَّل هنا أولًا. أسماء الحقول في JSON بصيغة camelCase (serde `rename_all = "camelCase"`).

## الأنواع

```ts
type IndexState = "idle" | "indexing" | "ready" | "error";

interface IndexStatus {
  state: IndexState;
  entries: number;          // عدد عناصر الفهرس المستخدم في البحث (0 أثناء أول فهرسة إن لم توجد لقطة)
  roots: string[];          // المجلدات المفهرسة
  elapsedMs: number | null; // مدة آخر فهرسة مكتملة؛ null إن كان الفهرس الحالي محمّلًا من لقطة
  error: string | null;     // رسالة إنجليزية عند state = "error"
}

interface ResultItem {
  id: number;               // معرّف نتيجة = (جيل العرض << 32) | رقم العنصر، دائمًا < 2^53 (آمن في JS).
                            // يُستخدم في أوامر الفتح كما هو (رقم معتم، لا تحلّله الواجهة). لا يُرسل مسار من الواجهة أبدًا
  name: string;             // الاسم الأصلي كما على القرص
  parent: string;           // المسار الكامل للمجلد الأب
  isDir: boolean;
  size: number | null;      // بالبايت؛ null للمجلدات
  modified: number | null;  // Unix seconds
  ext: string;              // الامتداد بحروف صغيرة بلا نقطة، "" إن لم يوجد
  highlights: [number, number][]; // مدى المطابقات داخل name بوحدات UTF-16 (مناسبة لـ JS string.slice)،
                            // مرتبة وغير متداخلة؛ تشمل الكلمات العادية وما طابقته الكلمات التقريبية (fuzzy)
  matched: "name" | "content" | "both"; // مصدر المطابقة (المهمة 4.7). "name" لكل نتائج بحث الأسماء وحده
  snippet: ContentSnippet | null;       // مقطع من المحتوى عند مطابقة المحتوى، وإلا null
}

interface ContentSnippet {
  text: string;             // مقطع قصير من النص الأصلي (بتشكيله)
  highlights: [number, number][]; // مدى الكلمات المطابقة داخل text بوحدات UTF-16، مرتبة وغير متداخلة
}

// حالة فهرسة المحتوى في الخلفية (المهمة 4.5)
interface ContentStatus {
  state: "disabled" | "idle" | "running" | "paused" | "stopped" | "error";
  pausedBy: ("user" | "battery" | "busy")[]; // أسباب الإيقاف المؤقت إن كانت state = "paused"
  processed: number;        // ما عولج من الطابور (مفهرس + محدّث + فاشل)
  total: number;            // processed + المتبقي في الطابور
  indexed: number;          // عدد المستندات في فهرس المحتوى
  failed: number;
  docsPerSecond: number;
  error: string | null;     // رسالة إنجليزية عند state = "error"
}

interface QueryError {
  message: string;          // إنجليزية
  kind: string;             // رمز ثابت من ParseErrorKind (مثل "UnknownFilter") لترجمته في الواجهة،
                            // أو "Unsupported": مرشّح صحيح لا يدعمه فهرس الأسماء بعد (content:, sem:, ~, tag:...)
  start: number;            // مدى الخطأ في نص الاستعلام بوحدات UTF-16
  end: number;
  suggestion: string | null;
}

type VolumeKind = "fixed" | "removable" | "network" | "unknown";

interface Volume {          // قرص في هذا الجهاز (list_volumes)
  path: string;             // جذر القرص: "C:\\" (ويندوز) أو "/" و "/media/usb" (غيره)
  name: string;             // اسم قصير للواجهة: "C:"؛ والمسار نفسه خارج ويندوز
  label: string | null;     // تسمية القرص ("Data") إن وُجدت
  kind: VolumeKind;         // fixed = داخلي (يُفهرس افتراضيًا)؛ removable/unknown لا يُفهرسان افتراضيًا
}

interface IndexRoots {      // إعداد "ماذا نفهرس" (get_index_roots / set_index_roots)
  wholeComputer: boolean;   // true = كل الجهاز (كل الأقراص الداخلية، يُعاد حسابها عند كل إقلاع)
  roots: string[];          // المجلدات التي يفهرسها هذا الإعداد الآن
}

interface SearchResponse {
  requestId: number;        // يُعاد كما أُرسل؛ الواجهة تتجاهل أي استجابة أقدم من آخر طلب
  items: ResultItem[];
  truncated: boolean;       // true إن وُجدت نتائج أكثر من limit
  elapsedMs: number;
  error: QueryError | null; // عند خطأ صياغة: items فارغة
  cancelled: boolean;       // true إن ألغاه طلب أحدث
  suggestion: SearchSuggestion | null; // "هل تقصد…": فقط حين items فارغة بلا خطأ وبلا إلغاء (انظر أدناه)
}

interface SearchSuggestion {
  query: string;            // نص الاستعلام البديل كاملًا (كما أُرسل، أي مع type: الذي تضيفه شريحة النوع)
  kind: "layout" | "fuzzy"; // layout: الكلمات أُعيدت كتابتها بلوحة المفاتيح الأخرى؛ fuzzy: أُضيف `fuzzy: ` في أوله
  count: number;            // عدد نتائجه (1..limit، مقصوص عند limit)
}

interface PreviewResponse {   // preview_entry
  requestId: number;        // يُعاد كما أُرسل؛ الواجهة تتجاهل أي استجابة أقدم من آخر طلب
  cancelled: boolean;       // true إن ألغاه طلب معاينة أحدث (preview = null)
  preview: Preview | null;
}

interface Preview {
  entry: { name: string; path: string; isDir: boolean; size: number | null; modified: number | null; ext: string };
  body: PreviewBody;
}

type PreviewBody =          // موسوم بالحقل kind
  | { kind: "text"; text: string; truncated: boolean; encoding: string | null }   // نص عادي (md، txt، csv، سجلات…)
  | { kind: "code"; text: string; truncated: boolean; encoding: string | null; language: string } // تلميح لغة للتلوين: "rust"، "python"، "json"…
  | { kind: "document"; text: string; truncated: boolean; format: string; title: string | null; author: string | null } // docx/xlsx/pptx/odt/ods/odp/epub/html/eml
  | { kind: "image"; src: string; mime: string } // عنوان athrix-preview لهذه المعاينة فقط (لا بايتات عبر IPC، ولا مسار)
  | { kind: "folder"; childCount: number; children: { name: string; isDir: boolean }[] } // أول 200 ابن: المجلدات أولًا ثم بالاسم
  | { kind: "unsupported"; reason: "pdf" | "format" | "tooLarge" | "encrypted" | "corrupt" | "missing" | "unreadable"; message: string };
```

### أجيال الفهرس ومعرّفات النتائج
- الفهرس حيّ (المهمة 1.5): أساس ثابت (نتيجة مشي أو لقطة) + طبقة تغييرات من مراقب الملفات. كل حالة تُنشر منه
  تسمى **عرضًا** (view) وتأخذ **جيلًا** جديدًا (21 بت، يلتف عند 2^21): بعد مشي، بعد تحميل لقطة، بعد كل دفعة تغييرات
  مطبّقة، وبعد كل دمج (compaction) للطبقة في أساس جديد.
- `ResultItem.id` يحمل الجيل ورقم العنصر: `generation * 2^32 + entry`. الشكل في JSON ما زال `number`. رقم العنصر
  لا معنى له إلا مع عرض جيله (عناصر الطبقة يُعاد ترقيمها مع كل دفعة)، لذا يُحلّ المعرّف دائمًا بعرض جيله نفسه.
- الخلفية تحتفظ بآخر **16 عرضًا** من الأساس الحالي: معرّف من نتائج ظهرت قبل بضع دفعات تغييرات ما زال يفتح ملفه.
  معرّف من عرض أقدم من ذلك، أو من قبل أساس جديد (إعادة فهرسة، تحميل لقطة، دمج)، يُرفض بالرسالة الثابتة
  `"this result is out of date; search again"` — ولا يُحلّ أبدًا إلى ملف آخر. على الواجهة عرض رسالة "أعد البحث" (أو إعادة البحث تلقائيًا).
- ملف حُذف بعد ظهوره في النتائج: معرّفه قد يبقى صالحًا (عرضه محفوظ)، فيفشل الفتح برسالة `"cannot open …"` من النظام.
- معرّف لا يمكن أن يصدر عن الخلفية (≥ 2^53) يُرفض بـ `"invalid result id …"`؛ وقبل جاهزية أي فهرس: `"the index is not ready yet"`.

### التحديث الحي (المهمة 1.5)
- عند بدء كل مشي يبدأ مراقب ملفات (`ReadDirectoryChangesW` على ويندوز) للمجلدات نفسها، بالاستثناءات نفسها، ويبقى
  يعمل بعد اكتمال المشي. التغييرات تُجمّع في دفعات (هدوء 300ms، وبحد أقصى 2s) وتُطبَّق على الفهرس خلال أقل من ثانية تقريبًا.
- التغييرات التي تحدث أثناء المشي تُطبَّق على نتيجته بعد اكتماله، فلا يضيع ملف أُنشئ أثناء فهرسة طويلة.
- إن أبلغ النظام عن فقد أحداث، تُطلب إعادة فهرسة للمجلدات نفسها عبر الطابور المعتاد (الأحدث يفوز): `indexing` ← `ready`.
- البحث لا ينتظر التحديث أبدًا: كل بحث يأخذ العرض الحالي مرة واحدة ويبني نتائجه ومعرّفاتها منه.
- الواجهة لا تحتاج أمرًا جديدًا. لتحديث النتائج المعروضة يمكنها إعادة البحث عند حدث `index://status` بعدد عناصر مختلف.

## نسخة واحدة من التطبيق
- تشغيل التطبيق مرة ثانية لا يفتح نسخة ثانية (لا أيقونة صينية ثانية ولا فشل في تسجيل الاختصار): النسخة الثانية تخرج فورًا
  وتُظهر النسخة الأولى نافذتها الرئيسية وتركّز عليها (`tauri-plugin-single-instance`).

## المعاينة (`preview_entry`، المهمتان 4.8 و 4.9)
- **النص والكود**: أول `maxChars` حرفًا (الافتراضي 65536، الحد الأعلى 1048576) عبر `athrix_extract::extract`، مع كشف
  الترميز (UTF-8/16، و windows-1256 و ISO-8859-6 للعربية). لا يُقرأ من الملف النصي إلا ما يكفي لهذا العدد مهما كان حجمه.
  `code` حين يُعرف للامتداد (أو لاسم مثل `Makefile`) تلميح لغة، وإلا `text`. الأسطر مفصولة بـ `\n`.
- **المستندات**: نص `athrix_extract` مع العنوان والمؤلف إن وُجدا؛ مستند أكبر من 32MB ⇠ `unsupported/tooLarge`.
  حدود الضغط (zip-bomb) من `Limits` مشدّدة للمعاينة.
- **PDF**: `unsupported/pdf` حاليًا؛ نقطة الربط `TODO(pdf-worker)` في `src-tauri/src/preview.rs` تنتظر عامل PDF المعزول.
- **الصور** (png، jpg/jpeg/jfif، gif، webp، bmp، ico، avif، svg): لا تمر بايتاتها عبر IPC. تسجّل الخلفية الملف تحت رمز
  عشوائي (128 بت) وتعيد `src` بمخطط URI خاص `athrix-preview` (`http://athrix-preview.localhost/<token>` على ويندوز،
  `athrix-preview://localhost/<token>` غيره). المعالج يخدم **فقط** آخر 8 صور عوينت، بالرمز (لا مسار في العنوان، والواجهة لا
  ترسل مسارًا أبدًا)، **للنافذة الرئيسية فقط** و GET فقط (غير ذلك ⇠ 403؛ رمز مجهول ⇠ 404)، بنوع MIME ثابت من قائمة
  بيضاء، مع `X-Content-Type-Options: nosniff` و `Cache-Control: no-store` و CSP مقفل (`default-src 'none'; sandbox`)،
  وحد 64MB. SVG يُعرض في `<img>` فقط (لا سكربتات). CSP التطبيق: `img-src` يشمل `athrix-preview:` و `http://athrix-preview.localhost`.
  أبعاد الصورة تقرؤها الواجهة من `naturalWidth/Height`.
- **المجلدات**: عدد الأبناء المباشرين من الفهرس (مسح للعرض كله مع فحص الإلغاء كل 64K عنصر) وأول 200 منهم.
- **التنفيذ**: على مجمّع المهام الحاجبة، معاينة واحدة في كل مرة (قفل)، والطلب الأحدث يلغي الأقدم: الملغى قبل بدئه لا يبدأ،
  ونتيجة الملغى بعد انتهائه تُهمل (`cancelled: true`). الاستخراج نفسه لا يُقطع لكنه محدود بـ `Limits`.
- **الإبراز داخل المعاينة (4.9)** يجري في الواجهة: كلمات الاستعلام العادية والعبارات وقيم `content:` (دون المنفية والمرشّحات)
  تُطبَّع (حالة الأحرف، التشكيل، التطويل، الهمزات، ى، الأرقام الهندية) وتُطابق في نص المعاينة مع خريطة إزاحات تعيد المدى
  إلى النص الأصلي بوحدات UTF-16. لا حاجة لأمر إضافي لكل ضغطة مفتاح.

## لقطة الفهرس عند الإقلاع (المهمة 1.4)
- تُحفظ في `<app_data>/index/names.snap` (مجلد بيانات التطبيق لـ `com.samt.athrix`) مع ملف جانبي
  `names.roots.json` يذكر المجلدات التي تغطيها. تُحفظ بعد كل مشي ناجح (في خيط الفهرسة، بعد تبديل الفهرس)،
  وتشمل التغييرات المراقَبة التي طُبّقت حتى لحظة الحفظ. لا تُحفظ بعد كل دفعة تغييرات ولا عند الخروج (مشي الإقلاع يصحّح الفرق).
- عند الإقلاع، إن وُجدت لقطة سليمة لنفس المجلدات بالضبط، تكون تسلسل الأحداث:
  `indexing (entries: 0)` ← `ready (entries: N, elapsedMs: null)` ← `indexing (entries: N)` ← `ready (entries: M, elapsedMs: …)`.
  البحث يعمل من حدث `ready` الأول؛ ومعرّفات نتائج اللقطة تصبح قديمة بعد `ready` الثاني.
- لقطة تالفة أو من إصدار آخر أو لمجلدات أخرى تُتجاهل (تُسجَّل في السجل فقط) ويجري مشي عادي: `indexing` ← `ready`.
- فشل حفظ اللقطة لا يظهر كخطأ فهرسة (يُسجَّل فقط).

## المجلدات المفهرسة (الإعداد الافتراضي: كل الجهاز)
- الافتراضي **كل الجهاز**: كل الأقراص الداخلية (`kind: "fixed"`) على ويندوز، مثل `["C:\\", "D:\\"]`؛ و `["/"]` خارج ويندوز.
  إن لم يوجد قرص داخلي ⇠ مجلد المستخدم. الأقراص القابلة للإزالة والشبكة لا تُفهرس إلا باختيارها.
- الإعداد يُحفظ في `<app_data>/indexed-roots.json` بالشكل `{"version":1,"roots":null}` (كل الجهاز) أو
  `{"version":1,"roots":["C:\\","D:\\"]}`. ملف مفقود أو تالف أو من إصدار آخر ⇠ كل الجهاز (يُسجَّل فقط).
- `IndexStatus.roots` يذكر دائمًا المجلدات الفعلية؛ الواجهة تعرضها في شريط الحالة وتبني منها محدّد نطاق البحث.
- لقطة الفهرس تُستخدم عند الإقلاع فقط إن طابقت مجلداتها المجلدات المحفوظة (انظر أعلاه)؛ تغيير الأقراص يعني مشيًا كاملًا.

## نطاق البحث (`search.scope`)
- الواجهة تعرض قبل الكتابة: «كل الجهاز» + قرص لكل مجلد في `IndexStatus.roots` (يُخفى المحدّد إن كان هناك مجلد واحد).
  الاختيار يُحفظ في التخزين المحلي (`athrix.searchScope`) ويُشارك بين النافذة الرئيسية والنافذة السريعة؛
  اختيار قرص لم يعد مفهرسًا يُعامل ككل الجهاز.
- التنفيذ في Rust: في الأساس يشغل كل جذر مفهرس كتلة معرّفات متصلة (الجذر ثم شجرته كاملة)، فيُفحص فقط ما يقع في
  النطاق، بالتوازي وبالترتيب نفسه وبالتوقف المبكر عند `limit + 1`، مع تخطي العناصر المحذوفة منذ المشي. عناصر طبقة
  التغييرات (المضافة أو المعدّلة بعد المشي، وهي قليلة) تُفحص كلها وتُصفّى بمسارها، وتأتي أولًا كما في البحث بلا نطاق.
  بعد الدمج تُلحق العناصر الجديدة بآخر الأساس فتصبح لجذرها كتل إضافية تُحسب تلقائيًا. لا يتغيّر نص الاستعلام.

## البحث التقريبي والترتيب (المهمة 3.7)
- `fuzzy:كلمة` مطابقة متسامحة (أخطاء إملائية، اختصارات `fuzzy:mrs` ⇠ "Monthly Report Summary"، أحرف متتابعة
  `fuzzy:rprt` ⇠ "report")؛ و `fuzzy:` وحدها (بلا قيمة) تجعل كل الكلمات العادية تقريبية.
- استعلام فيه كلمات تقريبية (خارج NOT) **مرتّب**: النتائج الأفضل أولًا (التعادل بترتيب الفهرس)، بدل ترتيب الفهرس. يسري
  ذلك على البحث بنطاق (قرص/مجلد) أيضًا: تُفحص كل عناصر النطاق ويُعاد أفضل `limit` منها، مع خلط عناصر طبقة التغييرات
  بعناصر الأساس حسب الدرجة. لا توقف مبكر في هذه الحالة.
- `highlights` تغطي الأحرف التي طابقتها الكلمة التقريبية (أحرف `rprt` في "report"، الأحرف الأولى للاختصار…).
  لا تغيير مطلوب في الواجهة غير ذلك.

## "هل تقصد…" (`SearchResponse.suggestion`، المهمة 3.3)
- يُحسب فقط حين يكون الاستعلام صحيحًا والفهرس جاهزًا ولم يجد شيئًا. وإلا `null`.
- المرشّحان بالترتيب، ويُعاد الأول الذي يجد نتائج (بالنطاق و `limit` وعلم الإلغاء نفسها، دون بناء صفوف):
  1. `layout`: كل كلمة حرة (عادية أو "عبارة"، والمنفية أيضًا) تُعاد كتابتها بلوحة المفاتيح الأخرى (العربية 101 ⇄ US
     QWERTY): `hgjrvdv ext:pdf` ⇠ `التقرير ext:pdf`، و `قثحخقف` ⇠ `report`. المرشّحات والعوامل والعلامات تبقى كما كُتبت.
  2. `fuzzy`: `fuzzy: ` + الاستعلام. لا يُقترح إن كان الاستعلام تقريبيًا أصلًا أو بلا كلمات عادية (مرشّحات فقط).
- الكلفة: بحث إضافي واحد أو اثنان فقط عند النتائج الصفرية؛ طلب أحدث يلغيهما كما يلغي البحث (`cancelled: true`).
- الواجهة تعرضه في حالة "لا نتائج": `layout` ⇠ زر «هل تقصد: التقرير؟»، `fuzzy` ⇠ زر «جرّب البحث التقريبي». الضغط يضع
  `query` في حقل البحث (بعد حذف لاحقة ` type:<kind>` التي أضافتها شريحة النوع، فالشريحة تبقى فعّالة).

## الاستثناءات الافتراضية للفهرسة
- بالاسم (أي مستوى، غير حساس لحالة الأحرف على ويندوز): `.git`، `node_modules`، `$RECYCLE.BIN`، `System Volume Information`،
  `$WinREAgent`، `$Windows.~BT`، `$Windows.~WS`، `$SysReset` (مجلدات تحديث/استعادة ويندوز المؤقتة).
- بالمسار: `**/AppData/Local/Temp`، `**/Windows/WinSxS` (مخزن مكونات ويندوز: نحو 100 ألف نسخة مرتبطة من ملفات النظام).
- خارج ويندوز فقط: `/proc`، `/sys`، `/dev`.
- لا تُتبع الروابط الرمزية ولا الـ junctions (مثل `Documents and Settings`). الملفات مثل `pagefile.sys` و `hiberfil.sys` تُفهرس كأسماء عادية.
